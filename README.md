    function renderAllProductGrids() {
      const categories = ['gung-nuong', 'gia-vi-muoi', 'hat-say', 'trai-cay-say', 'tra-ca-phe'];
      const searchLower = currentSearchTerm.trim().toLowerCase();

      categories.forEach((cat) => {
        const sectionEl = document.querySelector(`[data-category-section="${cat}"]`);
        const gridEl = document.getElementById(`grid-${cat}`);
        if (!sectionEl || !gridEl) return;

        // Kiểm tra lọc theo danh mục tổng
        if (currentCategory !== 'all' && currentCategory !== cat) {
          sectionEl.classList.add('hidden');
          return;
        }

        // Lọc sản phẩm thuộc nhóm (bao gồm sản phẩm cross-category nếu có)
        let list = PRODUCTS.filter((p) => p.category === cat || (p.extraCategories && p.extraCategories.includes(cat)));

        // Lọc theo danh mục con cho Hạt sấy & Trái cây sấy
        if (currentSubFilters[cat] && currentSubFilters[cat] !== 'all') {
          const sub = currentSubFilters[cat];
          list = list.filter((p) => p.subCategory === sub);
        }

        // Lọc theo từ khóa tìm kiếm
        if (searchLower) {
          list = list.filter(
            (p) =>
              p.name.toLowerCase().includes(searchLower) ||
              p.categoryLabel.toLowerCase().includes(searchLower) ||
              p.subCategory.toLowerCase().includes(searchLower)
          );
        }

        sectionEl.classList.remove('hidden');

        if (list.length === 0) {
          if (searchLower) {
            sectionEl.classList.add('hidden');
          } else {
            gridEl.innerHTML = `
              <div class="col-span-full bg-lk-cream border border-dashed border-gray-300 rounded-xl p-6 text-center space-y-2">
                <p class="text-sm font-semibold text-gray-700">Sản phẩm thuộc nhóm "${currentSubFilters[cat]}" đang được cập nhật hình ảnh mới.</p>

    function addToCart(productId) {
      const product = PRODUCTS.find((p) => p.id === productId);
      if (!product) return;
      const existing = cart.find((item) => item.id === productId);
      if (existing) {
        existing.qty += 1;
      } else {
        cart.push({ ...product, qty: 1 });
      }
      renderCart();
      showToast(`Đã thêm "${product.name}" vào giỏ hàng`);
    }

    function updateCartQty(productId, delta) {
      const item = cart.find((i) => i.id === productId);
      if (!item) return;
      item.qty += delta;
      if (item.qty <= 0) {
        cart = cart.filter((i) => i.id !== productId);
      }
      renderCart();
    }

    function renderCart() {
      const countBadge = document.getElementById('cartCountBadge');
      const listEl = document.getElementById('cartItemsList');
      const totalEl = document.getElementById('cartTotalAmount');

      const totalQty = cart.reduce((sum, i) => sum + i.qty, 0);
      const totalPrice = cart.reduce((sum, i) => sum + i.price * i.qty, 0);

      countBadge.textContent = totalQty;
      totalEl.textContent = formatVND(totalPrice);

      if (cart.length === 0) {
        listEl.innerHTML = `
          <div class="text-center py-10 space-y-2 text-gray-500">
            <p class="text-sm font-medium">Giỏ hàng của bạn đang trống</p>

          </div>
        `;
        return;
      }

      listEl.innerHTML = cart
        .map(
          (item) => `
          <div class="flex gap-3 p-2.5 rounded-xl border border-gray-100 bg-white">
            <img
              src="${driveImg(item.driveId)}"
              data-fallback="${driveFallback(item.driveId)}"

    // 10. TỰ ĐỘNG ĐỒNG BỘ MÀU CHỦ ĐẠO TỪ LOGO (NẾU TRÌNH DUYỆT CHO PHÉP CORS)

    function trySyncThemeFromLogo() {
      try {
        const probe = new Image();
        probe.crossOrigin = 'anonymous';
        probe.src = 'https://lh3.googleusercontent.com/d/1reuQPFzhcZFYfvFoniL03fHlw0xYjbMa=w120';
        probe.onload = function () {
          try {
            const canvas = document.createElement('canvas');
            const ctx = canvas.getContext('2d');
            canvas.width = 48;
            canvas.height = 48;
            ctx.drawImage(probe, 0, 0, 48, 48);
            const data = ctx.getImageData(0, 0, 48, 48).data;
            let rSum = 0, gSum = 0, bSum = 0, count = 0;

            for (let i = 0; i < data.length; i += 4) {
              const r = data[i], g = data[i + 1], b = data[i + 2], a = data[i + 3];
              if (a < 180) continue;
              const max = Math.max(r, g, b);
              const min = Math.min(r, g, b);
              const sat = max === 0 ? 0 : (max - min) / max;
              // Lọc bỏ màu trắng, đen, xám nhạt để lấy đúng màu thương hiệu trên logo
              if (sat > 0.28 && max > 45 && min < 220) {
                rSum += r;
                gSum += g;
                bSum += b;
                count++;
              }
            }

            if (count > 25) {
              const r = Math.round(rSum / count);
              const g = Math.round(gSum / count);
              const b = Math.round(bSum / count);
              const primaryHex = `rgb(${r}, ${g}, ${b})`;
              const darkHex = `rgb(${Math.max(0, Math.round(r * 0.72))}, ${Math.max(0, Math.round(g * 0.72))}, ${Math.max(0, Math.round(b * 0.72))})`;
              const softHex = `rgba(${r}, ${g}, ${b}, 0.08)`;
              document.documentElement.style.setProperty('--lk-primary', primaryHex);
              document.documentElement.style.setProperty('--lk-primary-dark', darkHex);
              document.documentElement.style.setProperty('--lk-primary-soft', softHex);
            }
          } catch (err) {

        goToBanner(currentBanner + 1);
        startBannerAutoSlide();

        dot.addEventListener('click', () => {
          goToBanner(Number(dot.dataset.slide));
          startBannerAutoSlide();
        });
      });

      const sliderEl = document.getElementById('heroSlider');
      let touchStartX = 0;
      sliderEl.addEventListener('touchstart', (e) => {
        touchStartX = e.changedTouches[0].screenX;
        stopBannerAutoSlide();
      }, { passive: true });
      sliderEl.addEventListener('touchend', (e) => {
        const diff = e.changedTouches[0].screenX - touchStartX;
        if (Math.abs(diff) > 40) {
          goToBanner(diff < 0 ? currentBanner + 1 : currentBanner - 1);
        }
        startBannerAutoSlide();
      }, { passive: true });

      // Sự kiện Menu Mobile
      document.getElementById('mobileMenuBtn').addEventListener('click', openMobileMenu);
      document.getElementById('closeMobileMenuBtn').addEventListener('click', closeMobileMenu);
      document.getElementById('mobileMenuOverlay').addEventListener('click', closeMobileMenu);

      // Sự kiện Tìm kiếm
      document.getElementById('searchToggleBtn').addEventListener('click', () => {
        const bar = document.getElementById('searchBarWrap');
        bar.classList.toggle('hidden');
        if (!bar.classList.contains('hidden')) {
          document.getElementById('productSearchInput').focus();
        }
      });

      const searchInput = document.getElementById('productSearchInput');
      const clearSearchBtn = document.getElementById('clearSearchBtn');
      searchInput.addEventListener('input', (e) => {
        currentSearchTerm = e.target.value;
        clearSearchBtn.classList.toggle('hidden', !currentSearchTerm.trim());
        currentCategory = 'all';
        renderAllProductGrids();
      });

        searchInput.value = '';
        currentSearchTerm = '';
        clearSearchBtn.classList.add('hidden');
        renderAllProductGrids();

      document.getElementById('cartOverlay').addEventListener('click', closeCart);

        const msgEl = document.getElementById('cartCheckoutMsg');
        if (cart.length === 0) {
          msgEl.className = 'text-xs bg-amber-50 text-amber-800 border border-amber-200 rounded-lg p-2.5';
          msgEl.textContent = 'Vui lòng chọn ít nhất 1 sản phẩm trước khi gửi đơn hàng.';
          msgEl.classList.remove('hidden');
          return;
        }
        const total = cart.reduce((s, i) => s + i.price * i.qty, 0);
        msgEl.className = 'text-xs bg-green-50 text-green-800 border border-green-200 rounded-lg p-2.5';
        msgEl.textContent = `Cảm ơn quý khách! Đơn hàng trị giá ${formatVND(total)} đã được ghi nhận. Nhân viên LK Đà Nẵng (0984.113.366) sẽ liên hệ xác nhận ngay.`;
        msgEl.classList.remove('hidden');
        cart = [];
        renderCart();
      });

      document.getElementById('detailModal').addEventListener('click', (e) => {
        if (e.target.id === 'detailModal') closeDetailModal();
      });

      document.getElementById('quoteForm').addEventListener('submit', (e) => {
        e.preventDefault();
        const name = document.getElementById('qName').value.trim();
        const phone = document.getElementById('qPhone').value.trim();
        const category = document.getElementById('qCategory').value;
        const statusEl = document.getElementById('quoteFormStatus');

        if (!name || !phone) {
          statusEl.className = 'rounded-lg px-4 py-3 text-xs sm:text-sm font-medium bg-red-50 text-red-700 border border-red-200';
          statusEl.textContent = 'Vui lòng điền đầy đủ Họ tên và Số điện thoại để LK Đà Nẵng gửi báo giá.';
          statusEl.classList.remove('hidden');
          return;
        }

        statusEl.className = 'rounded-lg px-4 py-3 text-xs sm:text-sm font-medium bg-green-50 text-green-800 border border-green-200';
        statusEl.textContent = `Đã tiếp nhận yêu cầu báo giá (${category}) của ${name}. Đội ngũ LK Đà Nẵng sẽ gọi/nhắn Zalo qua số ${phone} trong ít phút!`;
        statusEl.classList.remove('hidden');
        e.target.reset();
      });
    });
